-- هل الشي يخص لاعب (شخصيتي أو غيري) أو أداة/سلاح؟ نتجاهله
local function isPlayerStuff(object)
    if object:FindFirstAncestorOfClass("Tool") or object:IsA("Tool") then
        return true
    end
    for _, p in ipairs(Players:GetPlayers()) do
        local ch = p.Character
        if ch and (object == ch or object:IsDescendantOf(ch)) then
            return true
        end
    end
    return false
end

local function hasKeyword(object)
    local name = object.Name:lower()
    if CONFIG.namePattern and name:match(CONFIG.namePattern) then
        return true
    end
    for _, k in ipairs(CONFIG.keywords) do
        if name:find(k:lower(), 1, true) then
            return true
        end
    end
    return false
end

local function isAnimal(object)
    return object:IsA("Model")
        and object:FindFirstChildOfClass("Humanoid") ~= nil
end

local function matches(object)
    if not (object:IsA("Model") or object:IsA("BasePart")) then
        return false
    end
    if object:IsA("Terrain") or isPlayerStuff(object) then
        return false
    end
    if hasKeyword(object) then
        return true
    end
    return CONFIG.detectAnimals and isAnimal(object)
end

local function getPart(object)
    if object:IsA("BasePart") then
        return object
    end
    -- نفضّل الهيت بوكس (وبعده الجسم) عشان التصويب يكون عليه بالضبط
    local hit = object:FindFirstChild("Hitbox", true)
    if hit and hit:IsA("BasePart") then
        return hit
    end
    local body = object:FindFirstChild("Body", true)
    if body and body:IsA("BasePart") then
        return body
    end
    return object.PrimaryPart or object:FindFirstChildWhichIsA("BasePart", true)
end

-- ===== الرادار =====
local radarOn = false
local markers = {}

local function clearMarkers()
    for object, marker in pairs(markers) do
        marker:Destroy()
        markers[object] = nil
    end
end

-- لو أبو الشي (Model) ملوّن أصلاً، ما نكرر له سهم
local function hasMarkedAncestor(object)
    local p = object.Parent
    while p and p ~= Workspace do
        if markers[p] then
            return true
        end
        p = p.Parent
    end
    return false
end

local function inRadarRange(object, root)
    if not root then
        return true
    end
    local part = getPart(object)
    return part ~= nil and (part.Position - root.Position).Magnitude <= CONFIG.radarRange
end

local function updateRadar()
    if not radarOn then
        clearMarkers()
        return
    end

    local count = 0
    local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")

    local function markDhab(object, isCatch)
        count += 1
        if not markers[object] and not hasMarkedAncestor(object) then
            local marker = Instance.new("Highlight")
            marker.Name = isCatch and "DhabbCatchESP" or "DhabbESP"
            marker.Adornee = object
            marker.FillColor = isCatch and Color3.fromRGB(0, 220, 100) or Color3.fromRGB(255, 140, 0)
            marker.OutlineColor = isCatch and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(255, 255, 0)
            marker.FillTransparency = 0.35
            marker.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            marker.Parent = gui
            markers[object] = marker

            local part = getPart(object)
            if CONFIG.showTags and part then
                local tag = Instance.new("BillboardGui")
                tag.Name = "DhabbTag"
                tag.Adornee = part
                tag.AlwaysOnTop = true
                tag.Size = UDim2.fromOffset(90, 44)
                tag.StudsOffsetWorldSpace = Vector3.new(0, 3, 0)
                tag.Parent = marker

                local lbl = Instance.new("TextLabel")
                lbl.Name = "Dist"
                lbl.Size = UDim2.fromScale(1, 1)
                lbl.BackgroundTransparency = 1
                lbl.TextColor3 = isCatch and Color3.fromRGB(0, 255, 120) or Color3.fromRGB(255, 255, 0)
                lbl.TextStrokeTransparency = 0
                lbl.TextSize = 16
                lbl.Font = Enum.Font.SourceSansBold
                lbl.Text = isCatch and "★ شيل" or "▼"
                lbl.Parent = tag
            end
        end
    end

    -- فحص مجلد الضبان البرية أولاً لسرعة الأداء
    local wild = Workspace:FindFirstChild("DhabWild")
    if wild then
        for _, object in ipairs(wild:GetChildren()) do
            if matches(object) and inRadarRange(object, root) then
                markDhab(object, false)
            end
        end
    else
        for _, object in ipairs(Workspace:GetDescendants()) do
            if matches(object) and inRadarRange(object, root) then
                markDhab(object, false)
            end
        end
    end

    -- فحص الضبان المصيودة الجاهزة للجمع
    local catches = Workspace:FindFirstChild("DhabCatches")
    if catches then
        for _, object in ipairs(catches:GetChildren()) do
            if object:IsA("Model") and inRadarRange(object, root) then
                markDhab(object, true)
            end
        end
    end

    for object, marker in pairs(markers) do
        if not object:IsDescendantOf(Workspace) or isPlayerStuff(object)
            or not inRadarRange(object, root) then
            marker:Destroy()
            markers[object] = nil
        end
    end

    status.Text = "الرادار: لقيت " .. count .. " هدف"
end

-- ===== أداة الفحص: تعرض أسماء الأشياء اللي حولك =====
local function scanNearby()
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then
        status.Text = "الفحص: الشخصية مو جاهزة"
        return
    end

    local found, seen = {}, {}
    for _, object in ipairs(Workspace:GetDescendants()) do
        if (object:IsA("Model") or object:IsA("BasePart"))
            and not object:IsA("Terrain")
            and not isPlayerStuff(object) then
            local part = getPart(object)
            if part then
                local d = (part.Position - root.Position).Magnitude
                -- نتجاهل الأجزاء الكبيرة (أرض/جدران) والأسماء العامة
                local big = part.Size.Magnitude > 40
                if d <= CONFIG.scanRadius and not big and not seen[object.Name] then
                    seen[object.Name] = true
                    table.insert(found, { name = object.Name, class = object.ClassName, d = d })
                end
            end
        end
    end

    table.sort(found, function(a, b) return a.d < b.d end)

    local lines = {}
    for i = 1, math.min(#found, 8) do
        local f = found[i]
        lines[#lines + 1] = string.format("%s (%s) %dm", f.name, f.class, f.d)
        print("[DhabbScan]", f.name, f.class, math.floor(f.d))
    end

    status.Text = #lines > 0
        and ("أقرب الأسماء:\n" .. table.concat(lines, "\n") .. "\nوقف جنب الضب واضغط Scan، وانسخ اسمه في الخانة.")
        or "الفحص: ما لقيت شي قريب"
end

-- ===== الصيد التلقائي =====
local huntOn = false

local function findNearest(root)
    local best, bestDist = nil, CONFIG.maxHuntRange
    local wild = Workspace:FindFirstChild("DhabWild")
    if wild then
        for _, object in ipairs(wild:GetChildren()) do
            if matches(object) then
                local part = getPart(object)
                if part then
                    local d = (part.Position - root.Position).Magnitude
                    if d < bestDist then
                        best, bestDist = object, d
                    end
                end
            end
        end
    end
    if not best then
        for _, object in ipairs(Workspace:GetDescendants()) do
            if matches(object) then
                local part = getPart(object)
                if part then
                    local d = (part.Position - root.Position).Magnitude
                    if d < bestDist then
                        best, bestDist = object, d
                    end
                end
            end
        end
    end
    return best, bestDist
end

local function interact(object)
    local prompt = object:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt and fireproximityprompt then
        fireproximityprompt(prompt)
        return true
    end
    local click = object:FindFirstChildWhichIsA("ClickDetector", true)
    if click and fireclickdetector then
        fireclickdetector(click)
        return true
    end
    return false
end

-- يجهّز السلاح: يمسكه لو كان في الشنطة
local function getTool(char, humanoid)
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        return tool
    end
    local backpack = player:FindFirstChildOfClass("Backpack")

    if tool then

end

-- يوجّه الشخصية والكاميرا للضب
-- نقطة التصويب: مكان الضب + تقدّم لو يتحرك + رفع يعوّض نزول الرصاصة
local aimPart = nil

local aimTrackPart, aimLastPos, aimVel = nil, nil, Vector3.zero

local function computeAimPoint(part, dt)
    local cam = Workspace.CurrentCamera
    local origin = cam and cam.CFrame.Position or part.Position
    local dist = (part.Position - origin).Magnitude

    -- نحسب سرعة الضب من حركته الفعلية ونمرّرها بنعومة، فإذا وقف فجأة يوقف التقدّم بسرعة
    if aimTrackPart ~= part then
        aimTrackPart, aimLastPos, aimVel = part, part.Position, Vector3.zero
    end
    if dt and dt > 0 then
        local newVel = (part.Position - aimLastPos) / dt
        aimVel = aimVel:Lerp(newVel, math.clamp(dt * 8, 0, 1))
    end
    aimLastPos = part.Position

    local lead = Vector3.zero
    if aimVel.Magnitude > CONFIG.minLeadSpeed then
        lead = aimVel * (dist / CONFIG.bulletSpeed)
        if lead.Magnitude > CONFIG.maxLead then
            lead = lead.Unit * CONFIG.maxLead
        end
    end
    local lift = Vector3.new(0, dist * CONFIG.dropFactor + CONFIG.aimHeight, 0)
    return part.Position + lead + lift
end

-- نثبّت الكاميرا على الضب كل فريم، لأن سكربت الكاميرا يرجّعها فيصير التصويب عشوائي
pcall(function() RunService:UnbindFromRenderStep("DhabbAim") end)
RunService:BindToRenderStep("DhabbAim", Enum.RenderPriority.Camera.Value + 1, function(dt)
    local part = aimPart
    if not part or not part:IsDescendantOf(Workspace) then
        return
    end
    local cam = Workspace.CurrentCamera
    if cam then
        cam.CFrame = CFrame.lookAt(cam.CFrame.Position, computeAimPoint(part, dt))
    end
end)

local function aimAt(root, part)
    aimPart = part
    local flat = Vector3.new(part.Position.X, root.Position.Y, part.Position.Z)
    root.CFrame = CFrame.lookAt(root.Position, flat)
end

    local hud = playerGui:FindFirstChild("DhabHUD")
    if hud then
        local shootBtn = hud:FindFirstChild("Shoot", true)
        if shootBtn and shootBtn:IsA("GuiButton") and isShown(shootBtn) then
            pcall(function() clickGui(shootBtn) end)
        end
    end

    local cam = Workspace.CurrentCamera
    local cx, cy = 0, 0
    if cam then
        cx, cy = cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2
    end

    if mouse1click then
        pcall(mouse1click)
    else
        pcall(function()
            local vim = game:GetService("VirtualInputManager")
            vim:SendMouseButtonEvent(cx, cy, 0, true, game, 0)
            task.wait(0.04)
            vim:SendMouseButtonEvent(cx, cy, 0, false, game, 0)
        end)
    end
end

local function huntStep()
    local char = player.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not (humanoid and root) or humanoid.Health <= 0 then
        status.Text = "الصيد: الشخصية مو جاهزة"
        return
    end

    local target, dist = findNearest(root)
    if not target then
        aimPart = nil
        status.Text = "الصيد: ما فيه ضب قريب"
        return
    end

    local part = getPart(target)
    local tool = getTool(char, humanoid)

    if dist > CONFIG.shootRange then
        aimPart = nil
        humanoid:MoveTo(part.Position)
        status.Text = string.format("الصيد: ألحق %s (%d م)", target.Name, dist)
        return
    end

    -- داخل المدى: وجّه وأطلق (والضب يهرب فنتبعه بنفس الوقت)
    aimAt(root, part)
    humanoid:MoveTo(part.Position)

    if tool then
        fireTool(tool)
        status.Text = string.format("الصيد: أطلق على %s (%d م)", target.Name, dist)
    elseif dist <= CONFIG.interactRange and interact(target) then
        status.Text = "الصيد: تفاعلت مع " .. target.Name
    else
        status.Text = "الصيد: ما معي سلاح في الشنطة"
    end
end

-- ===== الجمع والتحميل: أزرار E / F (ProximityPrompt) =====
-- fireproximityprompt يضغط الزر مباشرة، فما يفرق E ولا F
local loadOn = false

local function promptMatches(prompt)
    local text = (prompt.ActionText .. " " .. prompt.ObjectText .. " "
        .. (prompt.Parent and prompt.Parent.Name or "")):lower()
    for _, k in ipairs(CONFIG.promptKeywords) do
        if text:find(k:lower(), 1, true) then
            return true
        end
    end
    return false
end

local function promptPosition(prompt)
    local p = prompt.Parent
    if not p then
        return nil
    end
    if p:IsA("BasePart") then
        return p.Position
    elseif p:IsA("Attachment") then
        return p.WorldPosition
    elseif p:IsA("Model") then
        return p:GetPivot().Position
    end
    local part = p:FindFirstAncestorWhichIsA("BasePart")
    return part and part.Position
end

-- يعرض الأزرار القريبة: الحرف، النص، اسم الشي، المسافة
local function scanPrompts()
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then
        status.Text = "Prompts: الشخصية مو جاهزة"
        return
    end

    local found = {}
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") and not isPlayerStuff(d) then
            local pos = promptPosition(d)
            if pos then
                local dist = (pos - root.Position).Magnitude
                if dist <= CONFIG.scanRadius then
                    table.insert(found, { p = d, dist = dist })
                end
            end
        end
    end
    table.sort(found, function(a, b) return a.dist < b.dist end)

    local lines = {}
    for i = 1, math.min(#found, 8) do
        local p = found[i].p
        local line = string.format("[%s] %s | %s | %s | %dm", p.KeyboardKeyCode.Name,
            p.ActionText, p.ObjectText, p.Parent and p.Parent.Name or "?", found[i].dist)
        lines[#lines + 1] = line
        print("[DhabbPrompt]", line)
    end
    status.Text = #lines > 0 and ("الأزرار القريبة:\n" .. table.concat(lines, "\n"))
        or "Prompts: ما فيه أزرار قريبة"
end

local function loadStep()
    if huntOn then
        return -- الصيد أول، بعدين التحميل
    end
    local char = player.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not (humanoid and root) or humanoid.Health <= 0 then
        status.Text = "التحميل: الشخصية مو جاهزة"
        return
    end

    local best, bestPos, bestDist = nil, nil, CONFIG.maxHuntRange
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Enabled and promptMatches(d) and not isPlayerStuff(d) then
            local pos = promptPosition(d)
            if pos then
                local dist = (pos - root.Position).Magnitude
                if dist < bestDist then
                    best, bestPos, bestDist = d, pos, dist
                end
            end
        end
    end

    if not best then
        status.Text = "التحميل: ما لقيت زر مطابق قريب"
    elseif bestDist > CONFIG.promptRange then
        humanoid:MoveTo(bestPos)
        status.Text = string.format("التحميل: أمشي لـ %s (%d م)", best.ActionText, bestDist)
    elseif fireproximityprompt then
        fireproximityprompt(best)
        status.Text = "التحميل: ضغطت [" .. best.KeyboardKeyCode.Name .. "] " .. best.ActionText
    else
        status.Text = "التحميل: المنفذ ما يدعم fireproximityprompt"
    end
end

task.spawn(function()
    while gui.Parent do
        task.wait(0.5)
        if loadOn then
            local ok, err = pcall(loadStep)
            if not ok then
                status.Text = "خطأ: " .. tostring(err)
            end
        end
    end
end)

-- تحرّك بالعصا/WASD والكاميرا تحدد الاتجاه: ارفع الكاميرا وامشي للأمام تطلع، نزّلها تنزل
local flyOn = false
local flyConn, flyBV

local function setFly(on)
    flyOn = on
    if flyConn then
        flyConn:Disconnect()
        flyConn = nil
    end
    if flyBV then
        flyBV:Destroy()
        flyBV = nil
    end

    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not on or not (hum and root) then
        flyOn = false
        setToggle(flyBtn, "Fly", false)
        return
    end

    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(1e5, 1e5, 1e5)
    flyBV.Velocity = Vector3.zero
    flyBV.Parent = root

    flyConn = RunService.Heartbeat:Connect(function()
        local cam = Workspace.CurrentCamera
        if not (flyBV and cam and hum.Parent) then
            return
        end
        local md = hum.MoveDirection
        local vel = Vector3.zero
        if md.Magnitude > 0.05 then
            local look = cam.CFrame.LookVector
            local flat = Vector3.new(look.X, 0, look.Z)
            local fwd = flat.Magnitude > 0.01 and md:Dot(flat.Unit) or 0
            vel = md.Unit * CONFIG.flySpeed + Vector3.new(0, look.Y * fwd * CONFIG.flySpeed, 0)
        end
        flyBV.Velocity = vel
    end)
end

flyBtn.Activated:Connect(function()
    setFly(not flyOn)
    setToggle(flyBtn, "Fly", flyOn)
end)

local function dumpAll()
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then
        status.Text = "Dump: الشخصية مو جاهزة"
        return
    end

    local out = { "== Dhabb dump ==", "radius: " .. CONFIG.dumpRadius, "", "-- أشياء قريبة --" }

    -- الأشياء القريبة (اسم / نوع / مسافة / المسار)
    local objs, seen = {}, {}
    for _, o in ipairs(Workspace:GetDescendants()) do
        if (o:IsA("Model") or o:IsA("BasePart")) and not o:IsA("Terrain") and not isPlayerStuff(o) then
            local part = getPart(o)
            if part and part.Size.Magnitude < 40 then
                local d = (part.Position - root.Position).Magnitude
                local key = o.Name .. o.ClassName
                if d <= CONFIG.dumpRadius and not seen[key] then
                    seen[key] = true
                    objs[#objs + 1] = { o = o, d = d }
                end
            end
        end
    end
    table.sort(objs, function(a, b) return a.d < b.d end)
    for i = 1, math.min(#objs, 80) do
        local e = objs[i]
        out[#out + 1] = string.format("%s | %s | %dm | %s", e.o.Name, e.o.ClassName, e.d, e.o:GetFullName())
    end

    -- الأزرار E / F القريبة
    out[#out + 1] = ""
    out[#out + 1] = "-- ProximityPrompts --"
    local n = 0
    for _, p in ipairs(Workspace:GetDescendants()) do
        if p:IsA("ProximityPrompt") and not isPlayerStuff(p) then
            local pos = promptPosition(p)
            if pos and (pos - root.Position).Magnitude <= CONFIG.dumpRadius and n < 40 then
                n += 1
                out[#out + 1] = string.format("[%s] %s | %s | %s", p.KeyboardKeyCode.Name,
                    p.ActionText, p.ObjectText, p:GetFullName())
            end
        end
    end

    -- أزرار الواجهة الظاهرة (مثل "حطه بالحوض")
    out[#out + 1] = ""
    out[#out + 1] = "-- Gui buttons --"
    local b = 0
    for _, g in ipairs(playerGui:GetDescendants()) do
        if (g:IsA("TextButton") or g:IsA("ImageButton")) and g.Visible
            and not g:IsDescendantOf(gui) and b < 60 then
            b += 1
            local label = g:IsA("TextButton") and g.Text or ""
            out[#out + 1] = string.format("%s | %s | %s", g.Name, label, g:GetFullName())
        end
    end

    local text = table.concat(out, "\n")
    print(text)

    local saved, copied = false, false
    if writefile then
        saved = pcall(writefile, "DhabbDump_" .. os.time() .. ".txt", text)
    end
    if setclipboard then
        copied = pcall(setclipboard, text)
    end

    status.Text = (saved and "حفظت ملف DhabbDump_رقم.txt في مجلد workspace للمنفذ. " or "")
        .. (copied and "ونسخته للحافظة، الصقه وأرسله لي." or (saved and "" or "المنفذ ما يدعم الحفظ، شوف النتيجة في الـ Console (F9)."))
end

-- ===== قراءة الماب كاملة (قراءة فقط، تحفظ ملف نصي) =====
local function fmtPos(v)
    return v and string.format("(%d,%d,%d)", v.X, v.Y, v.Z) or "(?)"
end

local function posOf(inst)
    if inst:IsA("BasePart") then
        return inst.Position
    elseif inst:IsA("ProximityPrompt") then
        return promptPosition(inst)
    elseif inst:IsA("Model") then
        local ok, cf = pcall(function() return inst:GetPivot() end)
        return ok and cf.Position or nil
    end
    return nil
end

local mapBusy = false

local function mapDump()
    if mapBusy then
        return
    end
    mapBusy = true
    status.Text = "الماب: أقرأ كل شي... ثواني"

    task.spawn(function()
        local out = { "== Dhabb MAP dump ==", "PlaceId: " .. tostring(game.PlaceId), "",
            "-- Workspace (المستوى الأول) --" }

        for _, c in ipairs(Workspace:GetChildren()) do
            out[#out + 1] = string.format("%s | %s | %d عنصر", c.Name, c.ClassName, #c:GetDescendants())
            task.wait()
        end

        local prompts, npcs, seats, hints = {}, {}, {}, {}
        local dhabCount, dhabSample = 0, {}
        local classCount = {}
        local hintWords = { "sell", "shop", "market", "town", "village", "store", "dealer", "garage",
            "بيع", "سوق", "ديرة", "متجر", "محل" }

        local all = Workspace:GetDescendants()
        for i, o in ipairs(all) do
            classCount[o.ClassName] = (classCount[o.ClassName] or 0) + 1

            if o:IsA("ProximityPrompt") and #prompts < 300 and not isPlayerStuff(o) then
                prompts[#prompts + 1] = string.format("[%s] %s | %s | %s | %s", o.KeyboardKeyCode.Name,
                    o.ActionText, o.ObjectText, fmtPos(promptPosition(o)), o:GetFullName())
            elseif o:IsA("Humanoid") and #npcs < 100 and o.Parent and not isPlayerStuff(o.Parent) then
                npcs[#npcs + 1] = string.format("%s | %s | %s", o.Parent.Name, fmtPos(posOf(o.Parent)),
                    o.Parent:GetFullName())
            elseif (o:IsA("VehicleSeat") or o:IsA("Seat")) and #seats < 80 then
                seats[#seats + 1] = string.format("%s | %s | %s | %s", o.ClassName, o.Name,
                    fmtPos(o.Position), o:GetFullName())
            end

            if (o:IsA("Model") or o:IsA("BasePart")) and not o:IsA("Terrain") then
                local lname = o.Name:lower()
                if CONFIG.namePattern and lname:match(CONFIG.namePattern) then
                    dhabCount += 1
                    if #dhabSample < 10 then
                        dhabSample[#dhabSample + 1] = o.Name .. " " .. fmtPos(posOf(o))
                    end
                elseif #hints < 80 then
                    for _, w in ipairs(hintWords) do
                        if lname:find(w, 1, true) then
                            hints[#hints + 1] = string.format("%s | %s | %s | %s", o.Name,
                                o.ClassName, fmtPos(posOf(o)), o:GetFullName())
                            break
                        end
                    end
                end
            end

            if i % 2500 == 0 then
                task.wait()
            end
        end

        out[#out + 1] = ""
        out[#out + 1] = "-- ضبان: العدد " .. dhabCount .. " (عينة) --"
        for _, s in ipairs(dhabSample) do out[#out + 1] = s end

        out[#out + 1] = ""
        out[#out + 1] = "-- أسماء فيها بيع/سوق/ديرة/محل (موقع) --"
        for _, s in ipairs(hints) do out[#out + 1] = s end

        out[#out + 1] = ""
        out[#out + 1] = "-- ProximityPrompts في كل الماب --"
        for _, s in ipairs(prompts) do out[#out + 1] = s end

        out[#out + 1] = ""
        out[#out + 1] = "-- شخصيات NPC --"
        for _, s in ipairs(npcs) do out[#out + 1] = s end

        out[#out + 1] = ""
        out[#out + 1] = "-- كراسي السيارات --"
        for _, s in ipairs(seats) do out[#out + 1] = s end

        out[#out + 1] = ""
        out[#out + 1] = "-- أعلى أنواع العناصر --"
        local classes = {}
        for k, v in pairs(classCount) do classes[#classes + 1] = { k, v } end
        table.sort(classes, function(a, b) return a[2] > b[2] end)
        for i = 1, math.min(#classes, 15) do
            out[#out + 1] = classes[i][1] .. ": " .. classes[i][2]
        end

        -- أسماء الـ Remotes (قراءة أسماء فقط)
        out[#out + 1] = ""
        out[#out + 1] = "-- ReplicatedStorage remotes (أسماء) --"
        local rs = game:GetService("ReplicatedStorage")
        local r = 0
        for _, o in ipairs(rs:GetDescendants()) do
            if (o:IsA("RemoteEvent") or o:IsA("RemoteFunction")) and r < 150 then
                r += 1
                out[#out + 1] = o.ClassName .. " | " .. o:GetFullName()
            end
        end

        local text = table.concat(out, "\n")
        local saved = false
        if writefile then
            saved = pcall(writefile, "DhabbMap_" .. os.time() .. ".txt", text)
        end
        for i = 1, #text, 8000 do
            print(text:sub(i, i + 7999))
        end

        status.Text = saved and "الماب: حفظت ملف DhabbMap_رقم.txt في مجلد workspace للمنفذ، أرسله لي."
            or "الماب: المنفذ ما يدعم الحفظ، النتيجة في الـ Console (F9)."
        mapBusy = false
    end)
end

local function getHumRoot()

-- سيارتي: اسمها يبدأ بـ DH_<UserId>_ ، وإلا أقرب سيارة لي
local function myCar()
    local cars = Workspace:FindFirstChild("PlayersCars")
    if not cars then
        return nil
    end
    local tag = "DH_" .. player.UserId .. "_"
    for _, c in ipairs(cars:GetChildren()) do
        if c.Name:sub(1, #tag) == tag then
            return c
        end
    end
    local _, root = getHumRoot()
    if not root then
        return nil
    end
    local best, bestD = nil, 80
    for _, c in ipairs(cars:GetChildren()) do
        local seat = c:FindFirstChild("DriverSeat")
        if seat then
            local d = (seat.Position - root.Position).Magnitude
            if d < bestD then
                best, bestD = c, d
            end
        end
    end
    return best
end

local function carPrompt(car, name)

        end
    end
    return nil
end

local function walkTo(pos, stopDist, timeout)
    local t0 = os.clock()
    local stuck, last = 0, nil
    while farmOn and os.clock() - t0 < timeout do
        local hum, root = getHumRoot()
        if not hum then
            return false
        end
        if (pos - root.Position).Magnitude <= stopDist then
            return true
        end
        hum:MoveTo(pos)
        if last and (root.Position - last).Magnitude < 0.3 then
            stuck += 0.25
            if stuck > 1.5 then
                hum.Jump = true
                stuck = 0
            end
        else
            stuck = 0
        end
        last = root.Position
        task.wait(0.25)
    end
    return false
end

-- طيران بدون تصادم: يطلع لارتفاع الرحلة، يمشي أفقياً، ثم ينزل على الهدف
local function flyTo(pos, stopDist, timeout, hover)
    local hum, root = getHumRoot()
    local char = player.Character
    if not hum or not char then
        return false
    end

    local saved = {}
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("BasePart") then
            saved[d] = d.CanCollide
            d.CanCollide = false
        end
    end
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
    bv.Velocity = Vector3.zero
    bv.Parent = root
    hum.PlatformStand = true

    local target = pos + Vector3.new(0, 3, 0)
    if hover then
        -- نبقى في الجو فوق النقطة (ما ننزل داخل الأرض)
        target = Vector3.new(pos.X, math.max(root.Position.Y, pos.Y) + 25, pos.Z)
    end
    local isShort = (target - root.Position).Magnitude <= 45
    local cruiseY = hover and target.Y or (isShort and (target.Y + 1) or (math.max(root.Position.Y, target.Y) + CONFIG.flyAltitude))
    local reached = false
    local t0 = os.clock()

    while farmOn and os.clock() - t0 < timeout do
        local h, r = getHumRoot()
        if not h then
            break
        end
        for part in pairs(saved) do
            if part.Parent then
                part.CanCollide = false
            end
        end

        local p = r.Position
        local flat = Vector3.new(target.X - p.X, 0, target.Z - p.Z)
        local to = target - p
        local vel
        if isShort then
            if to.Magnitude <= stopDist then
                reached = true
                break
            end
            vel = to.Unit * math.min(CONFIG.flySpeed, to.Magnitude * 4 + 4)
        elseif flat.Magnitude > stopDist + 20 then
            local vy = math.clamp((cruiseY - p.Y) * 2, -CONFIG.flySpeed, CONFIG.flySpeed)
            local hv = flat.Unit * CONFIG.flySpeed
            if p.Y < cruiseY - 10 then
                hv = hv * 0.25
            end
            vel = hv + Vector3.new(0, vy, 0)
        else
            if to.Magnitude <= stopDist then
                reached = true
                break
            end
            vel = to.Unit * math.min(CONFIG.flySpeed, to.Magnitude * 3 + 4)
        end
        bv.Velocity = vel
        task.wait(0.05)
    end

    bv:Destroy()
    pcall(function() hum.PlatformStand = false end)
    for part, v in pairs(saved) do
        if part.Parent then
            part.CanCollide = v
        end
    end
    return reached
end

local function travel(pos, stopDist, timeout)
    if CONFIG.flyTravel then
        return flyTo(pos, stopDist, timeout)
    end
    return walkTo(pos, stopDist, timeout)
end

-- يمشي/يطير لين يصير قريب من الزر ثم يضغطه
local function usePrompt(prompt)
    local pos = promptPosition(prompt)
    if not pos or not fireproximityprompt then
        return false
    end
    local reach = math.max(3, math.min(prompt.MaxActivationDistance - 1.5, CONFIG.promptRange))
    if not travel(pos, reach, 120) then
        return false
    end
    fireproximityprompt(prompt)
    return true
end

local function isShown(g)
    local cur = g
    while cur and cur ~= playerGui do
        if cur:IsA("GuiObject") and not cur.Visible then
            return false
        end
        if cur:IsA("ScreenGui") and not cur.Enabled then
            return false
        end
        cur = cur.Parent
    end
    return true
end

local function findButton(root, test)
    for _, g in ipairs(root:GetDescendants()) do
        if g:IsA("GuiButton") and isShown(g) and test(g) then

        end
    end
    return nil
end

end

-- ===== الجمع =====
local function pathHas(path, list)
    for _, w in ipairs(list) do
        if path:find(w, 1, true) then
            return true
        end
    end
    return false
end

local function isLootPrompt(p)
    if not p.Enabled then
        return false
    end
    if lootSkip[p] and os.clock() < lootSkip[p] then
        return false
    end
    -- الزر اللي طلع في فحصك على الضب المصيود: DhabCatches.<ضب>.Hitbox.CollectCatch
    if p.Name == "CollectCatch" then
        return true
    end
    local path = p:GetFullName()
    if pathHas(path, CONFIG.lootPaths) then
        return true
    end
    if pathHas(path, CONFIG.skipPaths) then
        return false
    end
    -- داخل موديل ضب (Dhab_xxxxxx)
    local m = p:FindFirstAncestorWhichIsA("Model")
    while m do
        if m.Name:lower():match(CONFIG.namePattern) then
            return true
        end
        m = m.Parent and m.Parent:FindFirstAncestorWhichIsA("Model") or nil
    end
    -- أو نص الزر يطابق كلمات الجمع/الذبح
    local text = (p.ActionText .. " " .. p.ObjectText):lower()
    for _, w in ipairs(CONFIG.pickupWords) do
        if text:find(w:lower(), 1, true) then
            return true
        end
    end
    return false
end

local function findLootPrompt(root)
    local best, bestD = nil, CONFIG.lootRadius
    local catches = Workspace:FindFirstChild("DhabCatches")
    if catches then
        for _, d in ipairs(catches:GetDescendants()) do
            if d:IsA("ProximityPrompt") and isLootPrompt(d) then
                local pos = promptPosition(d)
                if pos then
                    local dist = (pos - root.Position).Magnitude
                    if dist < bestD then
                        best, bestD = d, dist
                    end
                end
            end
        end
    end
    if not best then
        for _, d in ipairs(Workspace:GetDescendants()) do
            if d:IsA("ProximityPrompt") and isLootPrompt(d) then
                local pos = promptPosition(d)
                if pos then
                    local dist = (pos - root.Position).Magnitude
                    if dist < bestD then
                        best, bestD = d, dist
                    end
                end
            end
        end
    end
    return best
end

local function enterCar(car)
    local hum = getHumRoot()
    if hum and hum.SeatPart then
        return true
    end
    local p = carPrompt(car, "DrivePrompt")
    if not p then
        return false
    end
    usePrompt(p)
    local t0 = os.clock()
    while farmOn and os.clock() - t0 < 4 do

        if h and h.SeatPart then
            return true
        end
        task.wait(0.2)

local function exitCar()
    releaseKeys()
    local hum = getHumRoot()
    if not hum or not hum.SeatPart then
        return true
    end
    setKey(KEY.back, true)  -- فرملة
    farmWait(0.8)
    setKey(KEY.back, false)
    local t0 = os.clock()
    while farmOn and os.clock() - t0 < 4 do
        local h = getHumRoot()
        if not h or not h.SeatPart then
            return true
        end
        h.Sit = false
        h.Jump = true
        task.wait(0.4)
    end
    return false
end

local function driveTo(target, stopDist, maxTime)
    local t0 = os.clock()
    local stuck, reverseUntil = 0, 0
    while farmOn and os.clock() - t0 < maxTime do
        local hum = getHumRoot()
        local seat = hum and hum.SeatPart
        if not seat then
            releaseKeys()
            return false, "نزلت من السيارة"
        end

        local pos = seat.Position
        local flat = Vector3.new(target.X - pos.X, 0, target.Z - pos.Z)
        local dist = flat.Magnitude
        if dist <= stopDist then
            releaseKeys()
            return true
        end

        -- اتجاه السيارة: من الكرسي، ولو تمشي قدام نستخدم اتجاه حركتها الفعلي
        local vel = seat.AssemblyLinearVelocity
        local vflat = Vector3.new(vel.X, 0, vel.Z)
        local look = seat.CFrame.LookVector
        local heading = Vector3.new(look.X, 0, look.Z)
        if vflat.Magnitude > 8 and vflat:Dot(heading) > 0 then
            heading = vflat
        end
        if heading.Magnitude < 0.01 then
            heading = Vector3.new(0, 0, -1)
        end
        heading = heading.Unit
        local dir = flat.Unit
        local cross = heading:Cross(dir).Y -- موجب: الهدف على يسارك

        farmStatus(string.format("أسوق (%d م باقي)", dist))

        if os.clock() < reverseUntil then
            setKey(KEY.fwd, false)
            setKey(KEY.back, true)
            setKey(KEY.left, cross < 0)
            setKey(KEY.right, cross >= 0)
        else
            setKey(KEY.back, false)
            setKey(KEY.fwd, true)
            setKey(KEY.left, cross > 0.12)
            setKey(KEY.right, cross < -0.12)
            if vflat.Magnitude < 2 then
                stuck += 0.1
            else
                stuck = 0
            end
            if stuck > 3 then
                reverseUntil = os.clock() + 1.6
                stuck = 0
            end
        end
        task.wait(0.1)
    end
    releaseKeys()
    return false, "انتهى وقت السواقة"
end

local function sellTrip(car)
    aimPart = nil
    local hub = hubSell()
    local hubPos = (hub and (hub:IsA("BasePart") and hub.Position or hub:GetPivot().Position)) or cachedHubSellPos
    if not hubPos then
        farmStatus("ما لقيت مكان السوق")
        farmWait(3)
        return
    end

    farmStatus("أركب السيارة")
    if not enterCar(car) then
        farmStatus("ما قدرت أركب السيارة")
        farmWait(3)
        return
    end

    local ok, why = driveTo(hubPos, CONFIG.parkDistance, CONFIG.driveTimeout)
    if not ok then
        farmStatus("السواقة للسوق وقفت: " .. tostring(why))
        exitCar()
        farmWait(3)
        return
    end
    exitCar()

    if sellAll() then
        carried = 0
        pendingLoad = 0
    end

    -- رجوع لمنطقة الصيد
    if farmOn and farmArea then
        car = myCar() or car
        farmStatus("أرجع لمنطقة الصيد")
        if enterCar(car) then
            local back, why2 = driveTo(farmArea, 35, CONFIG.driveTimeout)
            if not back then
                farmStatus("الرجوع وقف: " .. tostring(why2))
            end
            exitCar()
        end
    end
end

-- ===== الصيد داخل المزرعة =====
local function farmHunt()
    local hum, root = getHumRoot()
    if not hum then
        farmStatus("الشخصية مو جاهزة")
        farmWait(1)
        return
    end

    -- أول شي: لو فيه ضب مصيود قريب نجمعه
    local loot = findLootPrompt(root)
    if loot then
        aimPart = nil
        farmStatus("أجمع: " .. loot.ActionText .. " " .. loot.ObjectText)
        local used = usePrompt(loot)
        farmWait(1.2)
        local gone = (not loot:IsDescendantOf(Workspace)) or (not loot.Enabled)
        if used and gone then
            lootTries[loot] = nil
            if CONFIG.useCar then
                pendingLoad += 1
            else
                carried += 1 -- بالخرج، ينباع مباشرة
            end
        else
            -- ما انشال: نعيد المحاولة مرتين وبعدها نتجاهله 30 ثانية
            lootTries[loot] = (lootTries[loot] or 0) + 1
            if lootTries[loot] >= 3 then
                lootSkip[loot] = os.clock() + 30
                lootTries[loot] = nil
            end
        end
        return
    end

    local target = findNearest(root)
    if target and farmArea then
        local part = getPart(target)
        if part and (part.Position - farmArea).Magnitude > CONFIG.farmRadius then
            target = nil
        end
    end

    if not target then
        aimPart = nil
        emptyHops += 1
        -- الماب يتحمّل حول اللاعب بس (StreamingEnabled)، فنتحرك عشان ينزل الضب
        if emptyHops > CONFIG.patrolHops and #CONFIG.huntGrounds > 0 then
            groundIdx = groundIdx % #CONFIG.huntGrounds + 1
            local g = CONFIG.huntGrounds[groundIdx]
            farmArea = g
            emptyHops = 0
            farmStatus("المنطقة فاضية، أطير لمنطقة ضبان ثانية")
            flyTo(g, 30, 150, true)
        else
            local base = farmArea or root.Position
            local ang = math.random() * math.pi * 2
            local r = math.random(40, math.max(60, math.floor(CONFIG.farmRadius * 0.8)))
            local dest = base + Vector3.new(math.cos(ang) * r, 0, math.sin(ang) * r)
            farmStatus("أدوّر على ضب حول المنطقة")
            flyTo(dest, 25, 45, true)
        end
        farmWait(1.5)
        return
    end

    emptyHops = 0
    huntStep()
end

local function farmLoop()
    if farmRunning then
        return
    end
    farmRunning = true
    while farmOn and gui.Parent do
        local ok, err = pcall(function()
            local hum, root = getHumRoot()
            if not hum then
                farmStatus("الشخصية مو جاهزة")
                farmWait(1)
                return
            end
            if not farmArea then
                farmArea = root.Position
            end

            local car = CONFIG.useCar and myCar() or nil
            if CONFIG.useCar and not car then
                farmStatus("ما لقيت سيارتك، طلّعها من قائمة Vehicles")
                farmWait(2)
                return
            end

            -- لو فيه ضبان بيدي، نحطها بالحوض
            if car and pendingLoad >= CONFIG.loadBatch then
                aimPart = nil
                local moved = loadIntoBed(car)
                carried += moved
                if moved == 0 then
                    farmStatus("ما لقيت شي أحطه بالحوض (أصفّر العداد)")
                end
                pendingLoad = 0
                return
            end

            if forceSell or carried >= CONFIG.bedCapacity then
                forceSell = false
                if car then
                    sellTrip(car)
                else
                    sellTripFly()
                end
                return
            end

            farmHunt()
            farmWait(CONFIG.huntInterval)
        end)
        if not ok then
            farmStatus("خطأ: " .. tostring(err))
            farmWait(1)
        end
    end
    releaseKeys()
    aimPart = nil
    farmRunning = false
end

local function weaponInfo()
    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then
        status.Text = "السلاح: الشخصية مو جاهزة"
        return

    local tool = getTool(char, hum)
    if not tool then
        status.Text = "السلاح: ما لقيت أداة في الشنطة"
        return
    end

    local lines = { "السلاح: " .. tool.Name }
    for _, d in ipairs(tool:GetDescendants()) do
        if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") or d:IsA("LocalScript")
            or d:IsA("Script") or d:IsA("ValueBase") then
            lines[#lines + 1] = d.ClassName .. ": " .. d.Name
            print("[DhabbWeapon]", d:GetFullName(), d.ClassName)
        end
    end
    for k, v in pairs(tool:GetAttributes()) do
        lines[#lines + 1] = "attr " .. k .. " = " .. tostring(v)
    end
    status.Text = table.concat(lines, "\n", 1, math.min(#lines, 9))
end

-- تعديل رفع التصويب من الواجهة
local function showAim()
    status.Text = string.format("رفع التصويب: %.2f لكل متر", CONFIG.dropFactor)
end
aimUpBtn.Activated:Connect(function()
    CONFIG.dropFactor += 0.01
    showAim()
end)
aimDownBtn.Activated:Connect(function()
    CONFIG.dropFactor = math.max(0, CONFIG.dropFactor - 0.01)
    showAim()
end)

-- إخفاء / إظهار الواجهة (الرادار والصيد يكملون شغل)
minBtn.Activated:Connect(function()
    mini.Position = frame.Position
    frame.Visible = false
    mini.Visible = true
end)
mini.Activated:Connect(function()
    frame.Position = mini.Position
    mini.Visible = false
    frame.Visible = true
end)

closeBtn.Activated:Connect(function()
    radarOn, huntOn = false, false
    aimPart = nil
    gui:Destroy()
end)

-- ===== الحلقات =====
task.spawn(function()
    while gui.Parent do
        task.wait(CONFIG.radarInterval)
        if radarOn then
            pcall(updateRadar)
        end
    end
end)

-- تحديث المسافة فوق كل ضب
task.spawn(function()
    while gui.Parent do
        task.wait(0.2)
        local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if radarOn and root then
            for object, marker in pairs(markers) do
                local part = object.Parent and getPart(object)
                local lbl = marker:FindFirstChild("DhabbTag")
                    and marker.DhabbTag:FindFirstChild("Dist")
                if part and lbl then
                    if marker.Name == "DhabbCatchESP" then
                        lbl.Text = string.format("★ شيل\n%d م", (part.Position - root.Position).Magnitude)
                    else
                        lbl.Text = string.format("▼\n%d م", (part.Position - root.Position).Magnitude)
                    end
                end
            end
        end
    end
end)

task.spawn(function()
    while gui.Parent do
        task.wait(CONFIG.huntInterval)
        if huntOn then
            local ok, err = pcall(huntStep)
            if not ok then
                status.Text = "خطأ: " .. tostring(err)
            end
        end
    end
end)

gui.Destroying:Connect(function()
    aimPart = nil
    setFly(false)
    setSpeed(false)
    pcall(function() RunService:UnbindFromRenderStep("DhabbAim") end)
    clearMarkers()
end)
